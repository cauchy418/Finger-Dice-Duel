// Run via execute_csharp_script in Play mode. Names, resources, raycast clicks and layout.
var boot=UnityEngine.Object.FindObjectOfType<FingerDiceDuel.UI.GameBootstrap>();
var flags=System.Reflection.BindingFlags.NonPublic|System.Reflection.BindingFlags.Public|System.Reflection.BindingFlags.Instance;
System.Func<object,string,object> field=(o,n)=>o.GetType().GetField(n,flags).GetValue(o);
System.Action<object,string,object[]> invoke=(o,n,a)=>o.GetType().GetMethod(n,flags).Invoke(o,a);
var output=System.IO.Path.GetFullPath("Art/Integration/four-role-emblems");System.IO.Directory.CreateDirectory(output);
var checks=new System.Collections.Generic.List<string>();
System.Action<bool,string> check=(ok,msg)=>{if(!ok)throw new System.Exception(msg);checks.Add("PASS "+msg);};
System.Action<UnityEngine.Canvas,string> shot=(canvas,name)=>{
 var go=new UnityEngine.GameObject("ProfessionReviewCamera",typeof(UnityEngine.Camera));var cam=go.GetComponent<UnityEngine.Camera>();cam.enabled=false;cam.orthographic=true;cam.clearFlags=UnityEngine.CameraClearFlags.SolidColor;
 var rt=new UnityEngine.RenderTexture(1920,1080,24);var pixels=new UnityEngine.Texture2D(1920,1080,UnityEngine.TextureFormat.RGB24,false);var prior=UnityEngine.RenderTexture.active;var mode=canvas.renderMode;var oldCam=canvas.worldCamera;var distance=canvas.planeDistance;
 try{cam.targetTexture=rt;canvas.renderMode=UnityEngine.RenderMode.ScreenSpaceCamera;canvas.worldCamera=cam;canvas.planeDistance=10;UnityEngine.Canvas.ForceUpdateCanvases();cam.Render();if(name!=null){UnityEngine.RenderTexture.active=rt;pixels.ReadPixels(new UnityEngine.Rect(0,0,1920,1080),0,0);pixels.Apply();System.IO.File.WriteAllBytes(System.IO.Path.Combine(output,name+".png"),UnityEngine.ImageConversion.EncodeToPNG(pixels));}}
 finally{canvas.renderMode=mode;canvas.worldCamera=oldCam;canvas.planeDistance=distance;cam.targetTexture=null;UnityEngine.RenderTexture.active=prior;UnityEngine.Object.DestroyImmediate(pixels);UnityEngine.Object.DestroyImmediate(rt);UnityEngine.Object.DestroyImmediate(go);UnityEngine.Canvas.ForceUpdateCanvases();}
};

boot.PlayerBIsAI=false;boot.AiThinkSeconds=1000;
System.Action<object,string,object> set=(o,n,v)=>o.GetType().GetField(n,flags).SetValue(o,v);
var front=((UnityEngine.RectTransform)field(boot,"_selectRoot")).GetComponentInParent<UnityEngine.Canvas>();
System.Func<UnityEngine.Canvas> battleCanvas=()=>((UnityEngine.RectTransform)field(field(boot,"_board"),"_canvasRoot")).GetComponent<UnityEngine.Canvas>();
System.Action<UnityEngine.UI.Button,UnityEngine.Canvas> click=(button,canvas)=>{
 shot(canvas,null);var data=new UnityEngine.EventSystems.PointerEventData(UnityEngine.EventSystems.EventSystem.current){position=UnityEngine.RectTransformUtility.WorldToScreenPoint(null,button.transform.position)};
 var hits=new System.Collections.Generic.List<UnityEngine.EventSystems.RaycastResult>();UnityEngine.EventSystems.EventSystem.current.RaycastAll(data,hits);
 check(button.gameObject.activeInHierarchy&&button.interactable&&hits.Count>0&&UnityEngine.EventSystems.ExecuteEvents.GetEventHandler<UnityEngine.EventSystems.IPointerClickHandler>(hits[0].gameObject)==button.gameObject,"raycast reaches "+button.name+" parent="+button.transform.parent.name+" active="+button.gameObject.activeInHierarchy+" enabled="+button.interactable+" phase="+((FingerDiceDuel.Assistant.BoardController)field(boot,"_controller"))?.Phase+" hits="+string.Join(",",hits.Select(h=>h.gameObject.name)));
 UnityEngine.EventSystems.ExecuteEvents.Execute(button.gameObject,data,UnityEngine.EventSystems.ExecuteEvents.pointerClickHandler);
};


var roles=new[]{FingerDiceDuel.Core.CharacterDefinition.DraconicBloodline,FingerDiceDuel.Core.CharacterDefinition.RangerHunter,FingerDiceDuel.Core.CharacterDefinition.WayOfTheOpenHand,FingerDiceDuel.Core.CharacterDefinition.WayOfTheFourElements};
check(FingerDiceDuel.Core.CharacterDefinition.BuiltIn.Length==37,"37 roles registered");
foreach(var def in FingerDiceDuel.Core.CharacterDefinition.BuiltIn)check(FingerDiceDuel.UI.BattleArt.Get("Subclasses/"+def.Id)!=null,"art "+def.Id);
invoke(boot,"GoToCharacterSelect",new object[0]);
var roster=(FingerDiceDuel.UI.CharacterRosterView)field(boot,"_rosterViewA");
var defs=(System.Collections.Generic.List<FingerDiceDuel.Core.CharacterDefinition>)field(roster,"_roster");
foreach(var def in roles){
 var sprite=FingerDiceDuel.UI.BattleArt.Get("Subclasses/"+def.Id);
 check(sprite.texture.width==512&&sprite.texture.height==512,"512 export "+def.Id);
 check(sprite==FingerDiceDuel.UI.BattleArt.Get("Classes/"+def.Identity.LegacyId),"legacy alias "+def.Id);
 int index=defs.IndexOf(def);check(index>=0,"selectable "+def.Id);roster.Select(index);roster.ScrollToSelected();
 var cards=(System.Collections.Generic.List<UnityEngine.UI.Button>)field(roster,"_cards");
 check(cards[index].transform.Find("Emblem").GetComponent<UnityEngine.UI.Image>().sprite==sprite,"selection painting "+def.Id);
}
shot(front,"selection-new-four");
invoke(boot,"OnCodexClicked",new object[0]);var codex=(FingerDiceDuel.UI.CodexScreen)field(boot,"_codex");check(codex.PageCount==37,"37 codex entries");
foreach(var def in roles){
 codex.GoTo(System.Array.IndexOf(FingerDiceDuel.Core.CharacterDefinition.BuiltIn,def));
 check(((UnityEngine.UI.Image)field(codex,"_emblem")).sprite==FingerDiceDuel.UI.BattleArt.Get("Subclasses/"+def.Id),"codex painting "+def.Id);shot(front,"codex-"+def.Id);
}
foreach(var profession in new[]{"sorcerer","ranger","monk"}){
 codex.ToggleProfession(profession);check(codex.PageCount==(profession=="monk"?4:2),"profession filter "+profession);codex.ToggleProfession(profession);
}
foreach(var def in roles){
 boot.StartMatch(def,FingerDiceDuel.Core.CharacterDefinition.Champion);invoke(boot,"RefreshView",new object[0]);
 var canvas=battleCanvas();var expected=FingerDiceDuel.UI.BattleArt.Get("Subclasses/"+def.Id);
 check(canvas.GetComponentsInChildren<UnityEngine.UI.Image>(true).Any(i=>i.sprite==expected),"battle painting "+def.Id);
 if(def==roles[0])shot(canvas,"battle-draconic");
}
var preview=System.AppDomain.CurrentDomain.GetAssemblies().Select(a=>a.GetType("FingerDiceDuel.EditorTools.MultiplayerArtPreview")).First(t=>t!=null);
preview.GetField("_boot",System.Reflection.BindingFlags.Static|System.Reflection.BindingFlags.NonPublic).SetValue(null,boot);
var build=preview.GetMethod("Build",System.Reflection.BindingFlags.Static|System.Reflection.BindingFlags.NonPublic);
foreach(int count in new[]{4,8}){
 build.Invoke(null,new object[]{count,0,roles[count==4?1:3],false});
 var expected=FingerDiceDuel.UI.BattleArt.Get("Subclasses/"+roles[count==4?1:3].Id);
 check(battleCanvas().GetComponentsInChildren<UnityEngine.UI.Image>(true).Any(i=>i.sprite==expected),"multiplayer painting "+count);
 shot(battleCanvas(),count+"p-new-emblems");
}
System.IO.File.WriteAllLines(System.IO.Path.Combine(output,"validation.txt"),checks);
boot.PlayerBIsAI=true;boot.ShowScreen(FingerDiceDuel.UI.GameScreen.MainMenu);return "PASS "+checks.Count+" checks";
