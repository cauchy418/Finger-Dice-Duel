powershell -NoProfile -ExecutionPolicy Bypass -File Tools/Statistics/run-current.ps1 -Run -Date 2026-11-27 -Directory 2026-11-27-light-dark-evocation-replay -Baseline "Tools/Statistics/2026-11-27-light-dark-evocation/baseline-results.json" -RuntimeVersion 8.0.15 -ProcessorAffinity 1
